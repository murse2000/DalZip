cmake_minimum_required(VERSION 3.20)
project(DalZipShell LANGUAGES CXX)
add_library(DalZipShell SHARED ShellExtension.cpp ShellExtension.def)
target_compile_features(DalZipShell PRIVATE cxx_std_17)
target_compile_definitions(DalZipShell PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN)
target_link_libraries(DalZipShell PRIVATE ole32 shell32 shlwapi advapi32 uuid)
if(MSVC)
  target_compile_options(DalZipShell PRIVATE /utf-8 /W4 /EHsc)
endif()
